```bash
#!/usr/bin/env bash
# Read-only heuristic for a hardcoded-values pass. Requires ripgrep. No JDK.
set -euo pipefail
cd "${1:-.}"
X=(--type java -g '!**/test/**' -g '!**/generated/**' -g '!ezkey-pam/**' -g '!ezkey-tests/**'
   -g '!experimental-hybrid/**' -g '!didactic/**' -g '!ezkey-demo*/**' -g '!ezkey-jenkins/**' -g '!ezkey-sdk/**')
echo "== raw counts (Java main, excluding PAM/tests)"
echo "numeric literals : $(rg "${X[@]}" -o '\b[0-9]+[LlDdFf]?\b' | wc -l)"
echo "excluding 0/1/2/-1: $(rg "${X[@]}" --pcre2 -o '(?<![\w.])([3-9]|[1-9][0-9]+)[Ll]?\b' | wc -l)"
echo "after noise filters: $(rg "${X[@]}" --pcre2 '(?<![\w."])([3-9]|[1-9][0-9]+)[Ll]?\b' \
  | rg -v '^\S+:\s*(//|\*|/\*|import|package|@)' \
  | rg -v 'static final|HttpStatus|\.get\(|\[\d+\]|log\.|LOG\.|logger\.|@Schema|@Size|@Max|@Min|@Column|@Operation|@ApiResponse|"' | wc -l)"
echo "== high-signal families"
echo "-- inline durations (excluding static final)"
rg "${X[@]}" -n -e 'plus(Seconds|Minutes|Hours|Days)\(\s*\d' -e 'Duration\.of(Seconds|Minutes|Hours|Days|Millis)\(\s*\d' | rg -v 'static final' || true
echo "-- @Scheduled without a \${...} placeholder"
rg "${X[@]}" -n '@Scheduled\((cron|fixed\w+)\s*=\s*"[^$]' || true
echo "-- named TTL/timeout/cap/batch constants"
rg "${X[@]}" -n -i 'static final (int|long|Duration)\s+\w*(TTL|EXPIR|TIMEOUT|WINDOW|RETENTION|MAX|LIMIT|BATCH|RETRY|DAYS|HOURS|MINUTES|SECONDS|MS)\w*\s*=' || true
echo "-- Java<->SQL: DEFAULT/INTERVAL/CHECK/VARCHAR in Flyway"
rg -n -i "interval '|DEFAULT \(|CHECK \(|VARCHAR\(\d{3,}\)" -g '**/db/migration/**/*.sql' || true
echo "-- orphan candidates: ezkey.* keys in properties with no Java reader (camelCase leaf; misses some, see A2)"
for k in $(rg -o -N --no-filename '^ezkey\.[a-z0-9.-]+' -g '*/config/*.properties' -g '**/application*.properties' | sort -u); do
  leaf=$(echo "$k" | awk -F. '{print $NF}')
  camel=$(echo "$leaf" | sed -E 's/-([a-z])/\U\1/g')
  rg -q --type java -g '!**/test/**' -e "$k" -e "\b$camel\b" . || echo "ORPHAN? $k"
done
echo "== Admin UI (excluding tests/generated/i18n)"
rg -n -e '\b(size|staleTime|max|timeout|interval|delay):?\s*\(?\s*[0-9_]{3,}' ezkey-admin-ui/src -g '*.ts' -g '*.tsx' \
  -g '!**/*.test.*' -g '!**/__tests__/**' -g '!**/generated/**' -g '!**/locales/**' || true
```
